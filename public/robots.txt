# Sọ̀rọ̀ Sókè — The People's Yorùbá Dictionary
#
# Without this file the host answers /robots.txt with index.html, and a
# crawler reads the markup as 97 unparseable directives.

User-agent: *
Allow: /

# The dictionary itself: several megabytes of JSON that render nothing on
# their own, and whose content is already reachable as normal pages.
#
# This used to be a much bigger deal than it looked. When every word lived at a
# hash route the only way to read one was to run the page's JavaScript, and that
# JavaScript fetches this directory - so a crawler that renders JS was blocked
# here and captured the page mid-load, showing "Loading the dictionary…". Both
# halves of that are gone: each word now has its own address under /yo/ and its
# own HTML file with the definitions already in it, and nothing has to be
# executed to read one. Keeping this closed costs a crawler nothing now.
Disallow: /data/

# Four measurement pages, not content. They exist to find which layer of the
# shell delays first paint on PageSpeed's mobile test; see
# public/perf-bisect/README.md. Disallowed as well as noindex'd because they
# are deliberately broken-looking copies of the front page's text, and a
# crawler reading them as duplicates of / would be a real cost for a question
# that should be answered in a week.
Disallow: /perf-bisect/
